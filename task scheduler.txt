class Solution {
    public int leastInterval(char[] tasks, int n) {
        int[] count = new int[26];

        for (char task : tasks) {
            count[task - 'A']++;
        }

        int max = 0;
        int maxCount = 0;

        for (int c : count) {
            if (c > max) {
                max = c;
                maxCount = 1;
            } else if (c == max) {
                maxCount++;
            }
        }

        int result = (max - 1) * (n + 1) + maxCount;

        return Math.max(result, tasks.length);
    }
}
